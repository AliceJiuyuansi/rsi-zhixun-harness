docker build --platform linux/amd64 --build-arg RSI_RELEASE=0.16 -f harness.Dockerfile -t zhixun-rsi-harness:0.16 .

docker save -o zhixun-rsi-harness-0.16.tar zhixun-rsi-harness:0.16
